// Scratch: writes a drift file with Kanizu's own writer (DriftFile::save), for RevitGeoSuite's reader test.
use kanizu_core::revit::gis::{self, Drift, DriftFile};
use std::path::Path;

#[allow(clippy::too_many_arguments)]
fn row(layer: &str, field: &str, parameter: &str, binding: &str, uid: &str, revit: Option<&str>, gdb: Option<&str>, base: Option<&str>, conflict: bool) -> Drift {
    Drift {
        revit_key: format!("8d7f3c1e-0000|{uid}|0"),
        unique_id: uid.into(),
        layer: layer.into(),
        field: field.into(),
        parameter: parameter.into(),
        guid: String::new(),
        binding: binding.into(),
        label: "北改札".into(),
        revit: revit.map(Into::into),
        gdb: gdb.map(Into::into),
        base: base.map(Into::into),
        conflict,
        gdb_category: None,
        gdb_restriction: None,
    }
}

/// A unit category row as Kanizu reviews it: the GDB's station code with the Revit category and restriction it comes from.
fn category_row(uid: &str, revit: &str, gdb: &str, base: &str, conflict: bool) -> Drift {
    let (category, restriction) = kanizu_core::station_categories::revit_category(gdb).expect("a station code");
    Drift {
        gdb_category: Some(category.into()),
        gdb_restriction: restriction.map(Into::into),
        ..row("unit", "category", "", "", uid, Some(revit), Some(gdb), Some(base), conflict)
    }
}

fn main() {
    let out = std::env::args().nth(1).expect("station folder");
    let file = DriftFile {
        schema: gis::DRIFT_SCHEMA.into(),
        version: gis::DRIFT_VERSION,
        model: "8d7f3c1e-0000".into(),
        applied_version: "revit/lumine-8d7f3c1e/20261110T000000Z-aaaaaaaa".into(),
        rows: vec![
            row("unit", "name", "", "", "u-1", Some("改札"), Some("北改札"), Some("改札"), false),
            row("unit", "alt_name", "", "", "u-1", None, Some("北"), None, false),
            row("unit", "restrict", "", "", "u-1", Some("employeesonly"), Some("private"), Some("employeesonly"), false),
            row("unit", "shop_hours", "kanizu_shop_hours", "type", "u-1", Some("9-21"), Some("10-20"), Some("9-21"), false),
            category_row("u-2", "retail", "walkway", "nonpublic", true),
            row("unit", "name", "", "", "u-3", Some("店舗"), None, Some("店舗"), false),
            row("level", "ordinal", "", "", "lvl", Some("1"), Some("2"), Some("1"), false),
            row("unit", "level_id", "", "", "u-1", Some("a"), Some("b"), Some("a"), false),
        ],
        not_placed: vec!["8d7f3c1e-0000|x|0".into()],
    };
    let path = file.save(Path::new(&out)).expect("saved");
    println!("{}", path.display());
}
